PYTHON := python
PIP := pip
PYTEST := pytest
DOCKER_IMAGE := taxi-route-recommender
DOCKER_TAG := 1.0.0

.DEFAULT_GOAL := help

# ------------------------------------------------------------------------------
# Help
# ------------------------------------------------------------------------------

help:
	@echo ""
	@echo "==============================================================="
	@echo " Taxi Route Recommender System"
	@echo "==============================================================="
	@echo ""
	@echo "Available Commands:"
	@echo ""
	@echo "make install           Install project dependencies"
	@echo "make test              Run all unit tests"
	@echo "make coverage          Run tests with coverage"
	@echo "make train             Train the model"
	@echo "make predict           Run inference"
	@echo "make evaluate          Evaluate the model"
	@echo "make preprocess        Run preprocessing"
	@echo "make pipeline          Execute complete pipeline"
	@echo "make figures           Generate figures"
	@echo "make report            Build report"
	@echo "make presentation      Build presentation"
	@echo "make clean             Remove temporary files"
	@echo "make docker-build      Build Docker image"
	@echo "make docker-run        Run Docker container"
	@echo "make lint              Run code formatting checks"
	@echo ""

# ------------------------------------------------------------------------------
# Installation
# ------------------------------------------------------------------------------

install:
	$(PIP) install --upgrade pip
	$(PIP) install -r requirements.txt

# ------------------------------------------------------------------------------
# Testing
# ------------------------------------------------------------------------------

test:
	$(PYTEST) tests/

coverage:
	$(PYTEST) --cov=src tests/

# ------------------------------------------------------------------------------
# Pipeline
# ------------------------------------------------------------------------------

preprocess:
	$(PYTHON) scripts/preprocess.py

train:
	$(PYTHON) scripts/train.py

predict:
	$(PYTHON) scripts/predict.py

evaluate:
	$(PYTHON) scripts/evaluate.py

pipeline:
	$(PYTHON) scripts/run_pipeline.py

# ------------------------------------------------------------------------------
# Reports
# ------------------------------------------------------------------------------

figures:
	$(PYTHON) scripts/generate_figures.py

report:
	$(PYTHON) scripts/build_report.py

presentation:
	$(PYTHON) scripts/build_presentation.py

# ------------------------------------------------------------------------------
# Docker
# ------------------------------------------------------------------------------

docker-build:
	docker build -t $(DOCKER_IMAGE):$(DOCKER_TAG) .

docker-run:
	docker run --rm $(DOCKER_IMAGE):$(DOCKER_TAG)

docker-compose-up:
	docker compose up --build

docker-compose-down:
	docker compose down

# ------------------------------------------------------------------------------
# Code Quality
# ------------------------------------------------------------------------------

lint:
	python -m compileall src scripts

# ------------------------------------------------------------------------------
# Cleanup
# ------------------------------------------------------------------------------

clean:
	@echo "Cleaning project..."
	rm -rf __pycache__
	rm -rf .pytest_cache
	rm -rf htmlcov
	rm -rf .coverage
	find . -name "*.pyc" -delete
	find . -name "__pycache__" -delete

# ------------------------------------------------------------------------------
# Research Workflow
# ------------------------------------------------------------------------------

reproduce:
	$(MAKE) preprocess
	$(MAKE) train
	$(MAKE) evaluate
	$(MAKE) figures
	$(MAKE) report
	$(MAKE) presentation

# ------------------------------------------------------------------------------
# Project Information
# ------------------------------------------------------------------------------

info:
	@echo ""
	@echo "Taxi Route Recommender System"
	@echo "Version : 1.0.0"
	@echo "Author  : Rohit Shetye"
	@echo "License : MIT"
